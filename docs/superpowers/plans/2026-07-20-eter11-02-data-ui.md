**Goal:** Komplet danych kart (13 problemów, kompetencje, talenty, mentorzy, postacie) plus grywalny interfejs hot-seat w ciemnym motywie sci-fi.

**Architecture:** Dane jako moduły TS z walidacją. UI w React czyta stan z hooka `useGame`, który opakowuje reducer z planu 1. Komponenty są bezstanowe — cały stan żyje w silniku.

**Tech Stack:** React 18, TypeScript, Tailwind 3, Vitest.

Wymaga ukończenia: `2026-07-20-eter11-01-engine.md`
Spec: `docs/superpowers/specs/2026-07-20-eter11-web-design.md`

## Global Constraints

- Treści kart po polsku, z polskimi znakami diakrytycznymi.
- Karty dopisane technicznie (nieobecne w oryginalnej instrukcji) mają `draft: true`.
- Każda karta ma emoji w polu `art` — brak plików graficznych w fazie 1.
- Kolory wyłącznie przez zmienne CSS z `src/styles/theme.css` — żadnych wartości hex w komponentach.
- Kontrast tekstu min. 4.5:1 (odbiorcą jest ośmiolatek).
- Komponenty w `src/ui/` nie importują niczego z `src/admin/` ani `src/firebase/`.

// src/ui/useGame.ts
import { useCallback, useState } from 'react';
import { createGame, reduce, DEFAULT_CONFIG } from '../engine/reducer';
import { buildDeck } from '../data/cards';
import { ALL_PROBLEMS } from '../data/problems';
import type { Action, GameState, RulesConfig } from '../engine/types';

export interface PlayerSetup {
  id: string;
  name: string;
  characterId: string;
}

/**
 * Opakowanie silnika gry dla React.
 *
 * Historia stanów służy funkcji cofania w trybie testowym panelu
 * administracyjnego — silnik jest czysty, więc wystarczy zachować
 * poprzednie stany, bez odtwarzania ruchów.
 */
export function useGame(
  players: PlayerSetup[],
  seed: number,
  config: RulesConfig = DEFAULT_CONFIG,
) {
  const [state, setState] = useState<GameState>(() =>
    createGame({ players, deck: buildDeck(), problems: ALL_PROBLEMS, seed, config }),
  );
  const [history, setHistory] = useState<GameState[]>([]);
  const [rejection, setRejection] = useState<string | null>(null);

  const dispatch = useCallback((action: Action) => {
    setState((current) => {
      const result = reduce(current, action);
      if (result.rejected) {
        setRejection(result.rejected);
        return current;
      }
      setRejection(null);
      setHistory((prev) => [...prev, current]);
      return result.state;
    });
  }, []);

  const undo = useCallback(() => {
    setHistory((prev) => {
      if (prev.length === 0) return prev;
      setState(prev[prev.length - 1]);
      setRejection(null);
      return prev.slice(0, -1);
    });
  }, []);

  const dismissRejection = useCallback(() => setRejection(null), []);

  return { state, dispatch, rejection, dismissRejection, history, undo };
}

import type { Card, MissionState, Problem, SlotKey } from '../../engine/types';

interface ProblemBoardProps {

/** Plansza jednego problemu: opis plus cztery sloty na karty. */

      </header>

          const placed = mission.played.filter(
            (p) => p.problemId === problem.id && p.slotKey === slot.key,
          );

              key={slot.key}

                highlighted ? 'border-accent bg-raised cursor-pointer' : '',

                {slotLabel(slot.key)}

              <div className="mt-auto flex flex-wrap gap-1 pt-2">
                {placed.map((p) => (
                  <CardView key={p.card.id} card={p.card} compact />
                ))}
              </div>
            </button>
          );
        })}
      </div>
    </section>
  );
}

import { useState } from 'react';
import { ALL_CHARACTERS } from '../../data/characters';
import type { PlayerSetup } from '../useGame';

interface SetupScreenProps {
  onStart: (players: PlayerSetup[]) => void;
}

interface Draft {
  name: string;
  characterId: string;
}

/** Ekran startowy: nazwy graczy i wybór postaci. Od 2 do 4 graczy. */
export function SetupScreen({ onStart }: SetupScreenProps) {
  const [drafts, setDrafts] = useState<Draft[]>([
    { name: '', characterId: ALL_CHARACTERS[0].id },
    { name: '', characterId: ALL_CHARACTERS[3].id },
  ]);

  const update = (index: number, patch: Partial<Draft>) => {
    setDrafts((prev) => prev.map((d, i) => (i === index ? { ...d, ...patch } : d)));
  };

  const addPlayer = () => {
    if (drafts.length >= 4) return;
    setDrafts((prev) => [...prev, { name: '', characterId: ALL_CHARACTERS[0].id }]);
  };

  const removePlayer = (index: number) => {
    if (drafts.length <= 2) return;
    setDrafts((prev) => prev.filter((_, i) => i !== index));
  };

  const namesFilled = drafts.every((d) => d.name.trim().length > 0);

  const start = () => {
    if (!namesFilled) return;
    onStart(
      drafts.map((d, i) => ({
        id: `player-${i + 1}`,
        name: d.name.trim(),
        characterId: d.characterId,
      })),
    );
  };

  return (
    <main className="mx-auto max-w-3xl px-4 py-10">
      <h1 className="text-4xl font-black tracking-tight text-accent">ETER11</h1>
      <p className="mt-1 text-lg text-ink-dim">Save the World</p>
      <p className="mt-4 max-w-prose text-sm leading-relaxed">
        Wspólnie rozwiązujecie problemy świata. Każdy dokłada coś od siebie,
        uczycie się od siebie nawzajem i odkrywacie, jakie kompetencje przyszłości
        są dla Was ważne.
      </p>

      <section className="mt-8 space-y-4" aria-label="Gracze">
        {drafts.map((draft, index) => (
          <div key={index} className="rounded-xl border border-edge bg-surface p-4">
            <div className="flex items-center justify-between">
              <h2 className="font-bold">Gracz {index + 1}</h2>
              {drafts.length > 2 && (
                <button
                  type="button"
                  onClick={() => removePlayer(index)}
                  className="text-xs text-danger underline"
                >
                  Usuń
                </button>
              )}
            </div>

            <label className="mt-3 block text-sm">
              <span className="text-ink-dim">Imię</span>
              <input
                value={draft.name}
                onChange={(e) => update(index, { name: e.target.value })}
                placeholder="Wpisz imię"
                className="mt-1 w-full rounded-lg border border-edge bg-bg px-3 py-2 text-ink"
              />
            </label>

            <fieldset className="mt-3">
              <legend className="text-sm text-ink-dim">Postać</legend>
              <div className="mt-2 grid gap-2 sm:grid-cols-2">
                {ALL_CHARACTERS.map((character) => (
                  <label
                    key={character.id}
                    className={[
                      'flex cursor-pointer items-start gap-2 rounded-lg border p-2 text-sm',
                      draft.characterId === character.id
                        ? 'border-accent bg-raised'
                        : 'border-edge',
                    ].join(' ')}
                  >
                    <input
                      type="radio"
                      name={`character-${index}`}
                      checked={draft.characterId === character.id}
                      onChange={() => update(index, { characterId: character.id })}
                      className="sr-only"
                    />
                    <span className="text-2xl" aria-hidden="true">{character.art}</span>
                    <span>
                      <span className="block font-semibold">{character.name}</span>
                      <span className="block text-xs text-ink-dim">{character.traits}</span>
                    </span>
                  </label>
                ))}
              </div>
            </fieldset>
          </div>
        ))}
      </section>

      <div className="mt-6 flex flex-wrap gap-3">
        {drafts.length < 4 && (
          <button
            type="button"
            onClick={addPlayer}
            className="rounded-lg border border-edge px-4 py-2 text-sm"
          >
            Dodaj gracza
          </button>
        )}
        <button
          type="button"
          onClick={start}
          disabled={!namesFilled}
          className="rounded-lg bg-accent px-6 py-2 font-bold text-bg disabled:opacity-40"
        >
          Zaczynamy misję
        </button>
      </div>
      {!namesFilled && (
        <p className="mt-2 text-xs text-ink-dim">Wpisz imię każdego gracza, żeby zacząć.</p>
      )}
    </main>
  );
}

// src/ui/screens/MissionScreen.tsx
import { useState } from 'react';
import { cardFitsSlot } from '../../engine/rules';
import type { Card, SlotKey } from '../../engine/types';
import { CardView } from '../components/CardView';
import { ProblemBoard } from '../components/ProblemBoard';
import type { useGame } from '../useGame';

interface MissionScreenProps {
  game: ReturnType<typeof useGame>;
}

/**
 * Ekran rozgrywki hot-seat.
 * Widoczna jest ręka wyłącznie aktywnego gracza — urządzenie wędruje między
 * graczami, więc karty pozostałych muszą pozostać zakryte.
 */
export function MissionScreen({ game }: MissionScreenProps) {
  const { state, dispatch, rejection, dismissRejection } = game;
  const [selected, setSelected] = useState<{ card: Card; fromMat: boolean } | null>(null);
  const [handRevealed, setHandRevealed] = useState(false);

  const mission = state.mission;
  if (!mission) return null;

  const activePlayer = state.players[state.activePlayerIndex];
  const matUsed = mission.matUsedBy.filter((id) => id === activePlayer.id).length;
  const canUseMat = matUsed < state.config.maxMatCardsPerMission;

  const playSelected = (problemId: string, slotKey: SlotKey) => {
    if (!selected) return;
    dispatch({
      type: 'PLAY_CARD',
      playerId: activePlayer.id,
      cardId: selected.card.id,
      slotKey,
      problemId,
      fromMat: selected.fromMat,
    });
    setSelected(null);
    setHandRevealed(false);
  };

  const endTurnAction = (action: 'PASS' | 'SWAP_HAND') => {
    dispatch({ type: action, playerId: activePlayer.id });
    setSelected(null);
    setHandRevealed(false);
  };

  return (
    <main className="mx-auto max-w-6xl px-4 py-6">
      <header className="mb-4 flex flex-wrap items-center justify-between gap-3">
        <div>
          <h1 className="text-2xl font-black text-accent">Misja {state.missionNumber}</h1>
          <p className="text-sm text-ink-dim">
            Runda {mission.round} z {state.config.roundsPerMission}
          </p>
        </div>
        <div className="text-right">
          <p className="text-sm text-ink-dim">Kolej gracza</p>
          <p className="text-xl font-bold">{activePlayer.name}</p>
        </div>
      </header>

      {rejection && (
        <div
          role="alert"
          className="mb-4 flex items-center justify-between rounded-lg border border-danger bg-raised px-4 py-2 text-sm"
        >
          <span>{rejection}</span>
          <button type="button" onClick={dismissRejection} className="ml-4 underline">
            OK
          </button>
        </div>
      )}

      <div className="space-y-4">
        {mission.problems.map((problem) => (
          <ProblemBoard
            key={problem.id}
            mission={mission}
            problem={problem}
            selectedCard={selected?.card ?? null}
            onSlotClick={playSelected}
            canPlayInSlot={cardFitsSlot}
          />
        ))}
      </div>

      <section className="mt-6 rounded-2xl border border-edge bg-surface p-4">
        <div className="flex flex-wrap items-center justify-between gap-3">
          <h2 className="font-bold">Ręka gracza {activePlayer.name}</h2>
          <button
            type="button"
            onClick={() => setHandRevealed((v) => !v)}
            className="rounded-lg border border-edge px-3 py-1 text-sm"
          >
            {handRevealed ? 'Ukryj karty' : 'Pokaż moje karty'}
          </button>
        </div>

        {handRevealed ? (
          <div className="mt-3 flex flex-wrap gap-2">
            {activePlayer.hand.map((card) => (
              <CardView
                key={card.id}
                card={card}
                selected={selected?.card.id === card.id && !selected.fromMat}
                onClick={() => setSelected({ card, fromMat: false })}
              />
            ))}
          </div>
        ) : (
          <p className="mt-3 text-sm text-ink-dim">
            Karty są zakryte. Podaj urządzenie graczowi {activePlayer.name} i naciśnij
            „Pokaż moje karty”.
          </p>
        )}

        {activePlayer.mat.length > 0 && handRevealed && (
          <div className="mt-4">
            <h3 className="text-sm font-bold">
              Twoja postać {canUseMat ? '(możesz użyć 1 karty)' : '(karta już użyta w tej misji)'}
            </h3>
            <div className="mt-2 flex flex-wrap gap-2">
              {activePlayer.mat.map((card) => (
                <CardView
                  key={card.id}
                  card={card}
                  compact
                  disabled={!canUseMat}
                  selected={selected?.card.id === card.id && selected.fromMat}
                  onClick={canUseMat ? () => setSelected({ card, fromMat: true }) : undefined}
                />
              ))}
            </div>
          </div>
        )}

        <div className="mt-4 flex flex-wrap gap-3">
          <button
            type="button"
            onClick={() => endTurnAction('PASS')}
            className="rounded-lg border border-edge px-4 py-2 text-sm"
          >
            Pasuję
          </button>
          <button
            type="button"
            onClick={() => endTurnAction('SWAP_HAND')}
            className="rounded-lg border border-edge px-4 py-2 text-sm"
          >
            Wymieniam karty
          </button>
        </div>
      </section>

      {state.log.length > 0 && (
        <section className="mt-6" aria-label="Przebieg misji">
          <h2 className="text-sm font-bold text-ink-dim">Co się wydarzyło</h2>
          <ul className="mt-2 space-y-1 text-xs text-ink-dim">
            {state.log.slice(-6).map((entry, index) => (
              <li key={index}>{entry}</li>
            ))}
          </ul>
        </section>
      )}
    </main>
  );
}

// src/ui/screens/SummaryScreen.tsx
import { useState } from 'react';
import type { Card } from '../../engine/types';
import { CardView } from '../components/CardView';
import type { useGame } from '../useGame';

interface SummaryScreenProps {
  game: ReturnType<typeof useGame>;
}

const COMPETENCE_CATEGORIES = ['psychological', 'digital', 'social'];

/**
 * Podsumowanie misji.
 * Każdy gracz może zabrać jedną ze swoich zagranych kart na kartę postaci —
 * także po porażce, zgodnie z zasadą „nawet przegrana uczy”.
 * Kompetencję można zamiast tego przekazać innemu graczowi.
 */
export function SummaryScreen({ game }: SummaryScreenProps) {
  const { state, dispatch, rejection, dismissRejection } = game;
  const [sharing, setSharing] = useState<Card | null>(null);

  const mission = state.mission;
  if (!mission) return null;

  const won = mission.phase === 'won';

  return (
    <main className="mx-auto max-w-4xl px-4 py-8">
      <h1 className={`text-3xl font-black ${won ? 'text-success' : 'text-danger'}`}>
        {won ? 'Problem rozwiązany!' : 'Tym razem problem wygrał'}
      </h1>

      <p className="mt-3 max-w-prose text-sm leading-relaxed">
        {won
          ? 'Udało się. Każdy z Was może zabrać jedną ze swoich kart na kartę postaci albo przekazać kompetencję innemu graczowi — wtedy dostaje dodatkową kartę doświadczenia.'
          : 'Problem trafia na stos nierozwiązanych, ale to nie koniec. Porozmawiajcie, jakich kompetencji zabrakło. Każdy, kto wyłożył kartę, i tak zabiera jedną na swoją postać — Wasze postacie właśnie się uczą. Do tego problemu możecie wrócić po dwóch kolejnych misjach.'}
      </p>

      {rejection && (
        <div role="alert" className="mt-4 rounded-lg border border-danger bg-raised px-4 py-2 text-sm">
          <span>{rejection}</span>
          <button type="button" onClick={dismissRejection} className="ml-4 underline">OK</button>
        </div>
      )}

      <section className="mt-6 space-y-5">
        {state.players.map((player) => {
          const plays = mission.played.filter((p) => p.playerId === player.id);
          const alreadyTook = mission.takenToMat.includes(player.id);
          if (plays.length === 0) {
            return (
              <div key={player.id} className="rounded-xl border border-edge bg-surface p-4">
                <h2 className="font-bold">{player.name}</h2>
                <p className="mt-1 text-sm text-ink-dim">
                  W tej misji nie wyłożył żadnej karty.
                </p>
              </div>
            );
          }

          return (
            <div key={player.id} className="rounded-xl border border-edge bg-surface p-4">
              <div className="flex flex-wrap items-center justify-between gap-2">
                <h2 className="font-bold">{player.name}</h2>
                <span className="text-xs text-ink-dim">
                  {alreadyTook ? 'Karta zabrana' : 'Wybierz jedną kartę'}
                </span>
              </div>

              <div className="mt-3 flex flex-wrap gap-3">
                {plays.map((play) => {
                  const shared = mission.sharedCardIds.includes(play.card.id);
                  const canShare =
                    !shared &&
                    COMPETENCE_CATEGORIES.includes(play.card.category) &&
                    state.players.length > 1;

                  return (
                    <div key={play.card.id} className="flex flex-col gap-1">
                      <CardView card={play.card} disabled={shared || alreadyTook} />
                      <button
                        type="button"
                        disabled={alreadyTook || shared}
                        onClick={() =>
                          dispatch({
                            type: 'TAKE_CARD_TO_MAT',
                            playerId: player.id,
                            cardId: play.card.id,
                          })
                        }
                        className="rounded border border-edge px-2 py-1 text-xs disabled:opacity-40"
                      >
                        Zabieram
                      </button>
                      {canShare && (
                        <button
                          type="button"
                          onClick={() => setSharing(play.card)}
                          className="rounded border border-accent px-2 py-1 text-xs text-accent"
                        >
                          Uczę kogoś
                        </button>
                      )}
                      {shared && (
                        <span className="text-center text-[10px] text-success">Przekazana</span>
                      )}
                    </div>
                  );
                })}
              </div>

              {sharing && plays.some((p) => p.card.id === sharing.id) && (
                <div className="mt-3 rounded-lg border border-accent bg-raised p-3">
                  <p className="text-sm">
                    Komu przekazujesz kartę <strong>{sharing.name}</strong>?
                  </p>
                  <div className="mt-2 flex flex-wrap gap-2">
                    {state.players
                      .filter((other) => other.id !== player.id)
                      .map((other) => (
                        <button
                          key={other.id}
                          type="button"
                          onClick={() => {
                            dispatch({
                              type: 'SHARE_CARD',
                              fromPlayerId: player.id,
                              toPlayerId: other.id,
                              cardId: sharing.id,
                            });
                            setSharing(null);
                          }}
                          className="rounded-lg bg-accent px-3 py-1 text-sm font-bold text-bg"
                        >
                          {other.name}
                        </button>
                      ))}
                    <button
                      type="button"
                      onClick={() => setSharing(null)}
                      className="rounded-lg border border-edge px-3 py-1 text-sm"
                    >
                      Anuluj
                    </button>
                  </div>
                </div>
              )}
            </div>
          );
        })}
      </section>

      <button
        type="button"
        onClick={() => dispatch({ type: 'END_MISSION_SUMMARY' })}
        className="mt-8 rounded-lg bg-accent px-6 py-3 font-bold text-bg"
      >
        Dalej
      </button>
    </main>
  );
}

// src/ui/GameApp.tsx
import { useState } from 'react';
import { FinaleScreen } from './screens/FinaleScreen';
import { MissionScreen } from './screens/MissionScreen';
import { SetupScreen } from './screens/SetupScreen';
import { SummaryScreen } from './screens/SummaryScreen';
import { useGame, type PlayerSetup } from './useGame';

/** Wewnętrzny komponent — hook useGame wymaga listy graczy przy montowaniu. */
function RunningGame({ players, seed, onRestart }: {
  players: PlayerSetup[];
  seed: number;
  onRestart: () => void;
}) {
  const game = useGame(players, seed);
  const { state, dispatch } = game;

  if (state.phase === 'finale') {
    return <FinaleScreen game={game} onRestart={onRestart} />;
  }

  if (state.phase === 'missionSummary') {
    return <SummaryScreen game={game} />;
  }

  if (state.phase === 'mission') {
    return <MissionScreen game={game} />;
  }

  // Faza setup między misjami — problem czeka na odkrycie.
  return (
    <main className="mx-auto max-w-2xl px-4 py-16 text-center">
      <h1 className="text-3xl font-black text-accent">
        {state.missionNumber === 0 ? 'Gotowi do pierwszej misji?' : 'Kolejna misja czeka'}
      </h1>
      <p className="mt-3 text-sm text-ink-dim">
        Rozwiązane problemy: {state.solvedProblems.length} · Pozostało kart problemów:{' '}
        {state.problemPile.length}
      </p>
      <button
        type="button"
        onClick={() => dispatch({ type: 'START_MISSION' })}
        className="mt-8 rounded-lg bg-accent px-8 py-4 text-lg font-bold text-bg"
      >
        Odkryj problem
      </button>
    </main>
  );
}

export function GameApp() {
  const [session, setSession] = useState<{ players: PlayerSetup[]; seed: number } | null>(null);

  if (!session) {
    return (
      <SetupScreen
        onStart={(players) =>
          // Ziarno z zegara — każda rozgrywka tasuje talię inaczej.
          setSession({ players, seed: Date.now() })
        }
      />
    );
  }

  return (
    <RunningGame
      key={session.seed}
      players={session.players}
      seed={session.seed}
      onRestart={() => setSession(null)}
    />
  );
}
